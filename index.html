Panduan Setup Google Spreadsheet (5 Menit)

Ikuti langkah ini agar data tersimpan permanen di Google Drive Anda

Gratis & Aman
  1. Buka sheets.new untuk membuat Google Spreadsheet baru. Beri nama misalnya Database Kesehatan Pribadi.
  2. Di dalam spreadsheet tersebut, klik menu Ekstensi (Extensions) → Apps Script.
  3. Hapus semua kode yang ada di editor Apps Script, lalu salin dan tempel (paste) kode di bawah ini:
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetVitals = getOrCreateSheet(ss, 'Catatan_Kesehatan', [
    'ID', 'Waktu', 'Sistolik', 'Diastolik', 'Gula_Darah', 'Tipe_Gula', 'Nadi', 'Berat_Badan', 'Suhu', 'Catatan'
  ]);
  const sheetMeds = getOrCreateSheet(ss, 'History_Obat', [
    'ID', 'Waktu', 'Nama_Obat', 'Dosis', 'Aturan_Minum', 'Status', 'Catatan'
  ]);

  const vitals = sheetToObjects(sheetVitals);
  const meds = sheetToObjects(sheetMeds);

  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    vitals: vitals,
    meds: meds
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action; // 'add', 'delete', 'sync_all'
    const type = payload.type;     // 'vital' atau 'med'

    const sheetVitals = getOrCreateSheet(ss, 'Catatan_Kesehatan', [
      'ID', 'Waktu', 'Sistolik', 'Diastolik', 'Gula_Darah', 'Tipe_Gula', 'Nadi', 'Berat_Badan', 'Suhu', 'Catatan'
    ]);
    const sheetMeds = getOrCreateSheet(ss, 'History_Obat', [
      'ID', 'Waktu', 'Nama_Obat', 'Dosis', 'Aturan_Minum', 'Status', 'Catatan'
    ]);

    if (action === 'add') {
      const d = payload.data;
      if (type === 'vital') {
        sheetVitals.appendRow([d.id, d.datetime, d.systolic, d.diastolic, d.sugar, d.sugarType, d.hr, d.weight, d.temp, d.notes]);
      } else if (type === 'med') {
        sheetMeds.appendRow([d.id, d.datetime, d.name, d.dosage, d.timing, d.status, d.notes]);
      }
    } else if (action === 'delete') {
      const sheet = type === 'vital' ? sheetVitals : sheetMeds;
      const rows = sheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(payload.id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
    } else if (action === 'sync_all') {
      // Menimpa data dengan data lokal saat migrasi awal
      if (payload.vitals && payload.vitals.length > 0) {
        sheetVitals.getRange(2, 1, Math.max(sheetVitals.getLastRow(), 1), 10).clearContent();
        payload.vitals.forEach(d => {
          sheetVitals.appendRow([d.id, d.datetime, d.systolic, d.diastolic, d.sugar, d.sugarType, d.hr, d.weight, d.temp, d.notes]);
        });
      }
      if (payload.meds && payload.meds.length > 0) {
        sheetMeds.getRange(2, 1, Math.max(sheetMeds.getLastRow(), 1), 7).clearContent();
        payload.meds.forEach(d => {
          sheetMeds.appendRow([d.id, d.datetime, d.name, d.dosage, d.timing, d.status, d.notes]);
        });
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
  }
  return sheet;
}

function sheetToObjects(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const result = [];
  for (let i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    if (sheet.getName() === 'Catatan_Kesehatan') {
      result.push({
        id: String(data[i][0]), datetime: data[i][1], systolic: data[i][2], diastolic: data[i][3],
        sugar: data[i][4], sugarType: data[i][5], hr: data[i][6], weight: data[i][7], temp: data[i][8], notes: data[i][9]
      });
    } else {
      result.push({
        id: String(data[i][0]), datetime: data[i][1], name: data[i][2], dosage: data[i][3],
        timing: data[i][4], status: data[i][5], notes: data[i][6]
      });
    }
  }
  return result;
}
